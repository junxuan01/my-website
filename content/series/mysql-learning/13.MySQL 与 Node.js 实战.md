# MySQL 与 Node.js 实战

Talk is cheap, show me the code. 本章我们将用最流行的 `mysql2` 库，从零实现一个简单的用户管理模块。这也是很多前端工程师转后端的第一课。

## 1. 为什么选 `mysql2`？
- **Promise 支持**：原生支持 async/await。
- **性能**：比旧版 `mysql` 库更快。
- **安全**：预编译语句（Prepared Statements）支持好。

## 2. 项目初始化

```bash
mkdir mysql-node-demo
cd mysql-node-demo
npm init -y
npm install mysql2 dotenv
```

创建一个 `.env` 文件存放配置：
```ini
DB_HOST=localhost
DB_USER=app_user
DB_PASS=password123
DB_NAME=my_blog
```

## 3. 封装数据库连接 (db.js)

我们不希望每次查询都新建连接，那样太慢。我们要用 **连接池 (Connection Pool)**。

```javascript
/* src/db.js */
const mysql = require('mysql2/promise');
require('dotenv').config();

// 创建连接池
const pool = mysql.createPool({
  host: process.env.DB_HOST,
  user: process.env.DB_USER,
  password: process.env.DB_PASS,
  database: process.env.DB_NAME,
  waitForConnections: true,
  connectionLimit: 10,  // 最多保持 10 个连接
  queueLimit: 0
});

// 测试连接
pool.getConnection()
  .then(conn => {
    console.log('✅ Database connected successfully');
    conn.release();
  })
  .catch(err => {
    console.error('❌ Database connection failed:', err);
  });

module.exports = pool;
```

## 4. 实现用户模型 (userModel.js)

```javascript
/* src/userModel.js */
const db = require('./db');

const User = {
  // 1. 查：获取所有用户
  async findAll() {
    const [rows] = await db.execute('SELECT id, username, email FROM users');
    return rows;
  },

  // 2. 查：根据 ID 获取
  async findById(id) {
    const [rows] = await db.execute(
      'SELECT * FROM users WHERE id = ?', 
      [id]
    );
    return rows[0]; // 返回第一条或 undefined
  },

  // 3. 增：创建用户
  async create(username, email, age) {
    // 这里的 result 包含了 insertId (新生成的 ID)
    const [result] = await db.execute(
      'INSERT INTO users (username, email, age) VALUES (?, ?, ?)',
      [username, email, age]
    );
    return result.insertId;
  },

  // 4. 改：更新信息
  async update(id, data) {
    // 动态拼接 SQL (简易版，生产环境可以用 query builder)
    const keys = Object.keys(data);
    const values = Object.values(data);
    
    // 生成 "username = ?, age = ?" 这样的字符串
    const setClause = keys.map(k => `${k} = ?`).join(', ');
    
    const sql = `UPDATE users SET ${setClause} WHERE id = ?`;
    // 参数数组末尾加上 id
    const params = [...values, id];

    const [result] = await db.execute(sql, params);
    return result.affectedRows > 0;
  },

  // 5. 事务示例：转账
  async transfer(fromId, toId, amount) {
    const conn = await db.getConnection(); // 只有事务需要手动拿连接
    try {
      await conn.beginTransaction(); // 开启事务

      // 扣钱
      await conn.execute(
        'UPDATE accounts SET balance = balance - ? WHERE user_id = ?',
        [amount, fromId]
      );
      
      // 加钱（模拟报错：如果 toId 不存在，这里可能抛错）
      await conn.execute(
        'UPDATE accounts SET balance = balance + ? WHERE user_id = ?',
        [amount, toId]
      );

      await conn.commit(); // 提交
      return true;
    } catch (error) {
      await conn.rollback(); // 出错回滚
      throw error;
    } finally {
      conn.release(); // 务必释放连接回池子！
    }
  }
};

module.exports = User;
```

## 5. 调用测试 (index.js)

```javascript
/* src/index.js */
const User = require('./userModel');

async function run() {
  try {
    // 新增
    const newId = await User.create('tom', 'tom@cat.com', 5);
    console.log('Created user ID:', newId);

    // 查询
    const user = await User.findById(newId);
    console.log('User found:', user);

    // 转账测试
    await User.transfer(1, 2, 100);
    console.log('Transfer success');
    
  } catch (err) {
    console.error('Error:', err);
  } finally {
    process.exit();
  }
}

run();
```
