### src/config/db.js

```javascript
const mysql = require('mysql2/promise');
require('dotenv').config();

// 创建连接池
const pool = mysql.createPool({
  host: process.env.DB_HOST,
  user: process.env.DB_USER,
  password: process.env.DB_PASSWORD,
  database: process.env.DB_NAME,
  waitForConnections: true,
  connectionLimit: 10,      // 最多 10 个连接
  queueLimit: 0,
  enableKeepAlive: true,
  keepAliveInitialDelay: 0
});

// 测试连接
pool.getConnection()
  .then(connection => {
    console.log('✅ Database connected');
    connection.release();
  })
  .catch(err => {
    console.error('❌ Database connection failed:', err.message);
    process.exit(1);
  });

module.exports = pool;
```

**关键点**：
- ✅ 使用连接池（不是单个连接）
- ✅ 使用 `mysql2/promise` 支持 async/await
- ✅ 环境变量管理敏感信息

## 第 3 步：用户模型

### src/models/user.js

```javascript
const db = require('../config/db');

class User {
  // 查询所有用户
  static async findAll() {
    const [rows] = await db.execute(
      'SELECT id, username, email, age, created_at FROM users'
    );
    return rows;
  }

  // 根据 ID 查询
  static async findById(id) {
    const [rows] = await db.execute(
      'SELECT id, username, email, age, created_at FROM users WHERE id = ?',
      [id]
    );
    return rows[0];
  }

  // 根据用户名查询
  static async findByUsername(username) {
    const [rows] = await db.execute(
      'SELECT * FROM users WHERE username = ?',
      [username]
    );
    return rows[0];
  }

  // 创建用户
  static async create(userData) {
    const { username, email, age } = userData;
    
    const [result] = await db.execute(
      'INSERT INTO users (username, email, age) VALUES (?, ?, ?)',
      [username, email, age]
    );
    
    return {
      id: result.insertId,
      username,
      email,
      age
    };
  }

  // 更新用户
  static async update(id, userData) {
    const fields = [];
    const values = [];
    
    // 动态构建 UPDATE 语句
    if (userData.username) {
      fields.push('username = ?');
      values.push(userData.username);
    }
    if (userData.email) {
      fields.push('email = ?');
      values.push(userData.email);
    }
    if (userData.age !== undefined) {
      fields.push('age = ?');
      values.push(userData.age);
    }
    
    if (fields.length === 0) {
      throw new Error('No fields to update');
    }
    
    values.push(id);
    
    const [result] = await db.execute(
      `UPDATE users SET ${fields.join(', ')} WHERE id = ?`,
      values
    );
    
    return result.affectedRows > 0;
  }

  // 删除用户
  static async delete(id) {
    const [result] = await db.execute(
      'DELETE FROM users WHERE id = ?',
      [id]
    );
    return result.affectedRows > 0;
  }

  // 分页查询
  static async findWithPagination(page = 1, limit = 10) {
    const offset = (page - 1) * limit;
    
    // 查询总数
    const [countRows] = await db.execute(
      'SELECT COUNT(*) as total FROM users'
    );
    const total = countRows[0].total;
    
    // 查询数据
    const [rows] = await db.execute(
      'SELECT id, username, email, age, created_at FROM users LIMIT ? OFFSET ?',
      [limit, offset]
    );
    
    return {
      data: rows,
      pagination: {
        page,
        limit,
        total,
        totalPages: Math.ceil(total / limit)
      }
    };
  }
}

module.exports = User;
```

## 第 4 步：RESTful API 路由

### src/routes/users.js

```javascript
const express = require('express');
const router = express.Router();
const User = require('../models/user');

// GET /api/users - 获取所有用户（支持分页）
router.get('/', async (req, res) => {
  try {
    const { page, limit } = req.query;
    
    if (page && limit) {
      const result = await User.findWithPagination(
        parseInt(page),
        parseInt(limit)
      );
      return res.json(result);
    }
    
    const users = await User.findAll();
    res.json({ data: users });
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});

// GET /api/users/:id - 获取单个用户
router.get('/:id', async (req, res) => {
  try {
    const user = await User.findById(req.params.id);
    
    if (!user) {
      return res.status(404).json({ error: 'User not found' });
    }
    
    res.json({ data: user });
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});

// POST /api/users - 创建用户
router.post('/', async (req, res) => {
  try {
    const { username, email, age } = req.body;
    
    // 验证
    if (!username || !email) {
      return res.status(400).json({ 
        error: 'Username and email are required' 
      });
    }
    
    // 检查是否已存在
    const existingUser = await User.findByUsername(username);
    if (existingUser) {
      return res.status(409).json({ 
        error: 'Username already exists' 
      });
    }
    
    const user = await User.create({ username, email, age });
    res.status(201).json({ data: user });
  } catch (error) {
    // 处理唯一约束错误
    if (error.code === 'ER_DUP_ENTRY') {
      return res.status(409).json({ 
        error: 'Username or email already exists' 
      });
    }
    res.status(500).json({ error: error.message });
  }
});

// PUT /api/users/:id - 更新用户
router.put('/:id', async (req, res) => {
  try {
    const updated = await User.update(req.params.id, req.body);
    
    if (!updated) {
      return res.status(404).json({ error: 'User not found' });
    }
    
    const user = await User.findById(req.params.id);
    res.json({ data: user });
  } catch (error) {
    if (error.code === 'ER_DUP_ENTRY') {
      return res.status(409).json({ 
        error: 'Username or email already exists' 
      });
    }
    res.status(500).json({ error: error.message });
  }
});

// DELETE /api/users/:id - 删除用户
router.delete('/:id', async (req, res) => {
  try {
    const deleted = await User.delete(req.params.id);
    
    if (!deleted) {
      return res.status(404).json({ error: 'User not found' });
    }
    
    res.status(204).send();
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});

module.exports = router;
```

  res.json({ status: 'ok' });
});

app.use((req, res) => {
  res.status(404).json({ error: 'Not found' });
});

// 错误处理
app.use((err, req, res, next) => {
  console.error(err.stack);
  res.status(500).json({ error: 'Something went wrong!' });

  console.log(`🚀 Server running on http://localhost:${PORT}`);
});

## 事务处理示例

### 添加转账功能

**src/models/account.js**：

```javascript
const db = require('../config/db');

class Account {
  static async transfer(fromId, toId, amount) {
    const connection = await db.getConnection();
    
    try {
      await connection.beginTransaction();
      
      // 检查余额
      const [rows] = await connection.execute(
        'SELECT balance FROM accounts WHERE user_id = ? FOR UPDATE',
        [fromId]
      );
      
      if (!rows[0] || rows[0].balance < amount) {
        throw new Error('Insufficient balance');
      }
      
      // 扣款
      await connection.execute(
        'UPDATE accounts SET balance = balance - ? WHERE user_id = ?',
        [amount, fromId]
      );
      
      // 加款
      await connection.execute(
        'UPDATE accounts SET balance = balance + ? WHERE user_id = ?',
        [amount, toId]
      );
      
      // 记录交易
      await connection.execute(
        'INSERT INTO transactions (from_id, to_id, amount) VALUES (?, ?, ?)',
        [fromId, toId, amount]
      );
      
      await connection.commit();
      return { success: true };
    } catch (error) {
      await connection.rollback();
      throw error;
    } finally {
      connection.release();
    }
  }
}

module.exports = Account;
```

## 最佳实践

### 1. 使用参数化查询（防止 SQL 注入）

```javascript
// ❌ 危险！
const sql = `SELECT * FROM users WHERE username = '${username}'`;

// ✅ 安全
const [rows] = await db.execute(
  'SELECT * FROM users WHERE username = ?',
  [username]
);
```

### 2. 错误处理

```javascript
try {
  const user = await User.create(userData);
  res.json({ data: user });
} catch (error) {
  // 处理特定错误
  if (error.code === 'ER_DUP_ENTRY') {
    return res.status(409).json({ error: 'Already exists' });
  }
  
  // 记录日志
  console.error('Create user error:', error);
  
  // 返回通用错误
  res.status(500).json({ error: 'Internal server error' });
}
```

### 3. 连接池管理

```javascript
// ✅ 普通查询用连接池
const [rows] = await db.execute('SELECT * FROM users');

// ✅ 事务需要手动获取连接
const connection = await db.getConnection();
try {
  await connection.beginTransaction();
  // ...
  await connection.commit();
} finally {
  connection.release();  // 一定要释放！
}
```

```javascript
// ❌ 不要硬编码
